  SELECT id INTO target_id
  FROM public.schedules_manualv2
  WHERE prof = NEW.name
    AND date = NEW.log_date

    SET start_time = NEW.time_in,
        end_time   = NEW.time_out
    WHERE id = target_id;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;
